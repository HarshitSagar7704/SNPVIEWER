library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity AD7814_READ is
    Port (
        clk       : in  STD_LOGIC;
        ad_dout   : in  STD_LOGIC;
        ad_cs     : out STD_LOGIC;
        ad_sclk   : out STD_LOGIC;
        temp_data : out STD_LOGIC_VECTOR(9 downto 0)
    );
end AD7814_READ;

architecture Behavioral of AD7814_READ is
    signal sclk_reg : STD_LOGIC := '1';
    signal cs_reg   : STD_LOGIC := '1';

    signal clk_count : integer range 0 to 24 := 0;
    signal bit_count : integer range 0 to 16 := 0;

    signal data_reg : STD_LOGIC_VECTOR(9 downto 0) := (others => '0');
    signal delay_count : integer range 0 to 49999 := 0;

begin
    ad_sclk   <= sclk_reg;
    ad_cs     <= cs_reg;
    temp_data <= data_reg;

    process(clk)
    begin
        if rising_edge(clk) then

            if cs_reg = '1' then
                sclk_reg <= '1';

                if delay_count = 49999 then
                    delay_count <= 0;
                    cs_reg <= '0';
                    bit_count <= 0;
                else
                    delay_count <= delay_count + 1;
                end if;

            else
                if clk_count = 24 then
                    clk_count <= 0;
                    sclk_reg <= not sclk_reg;

                    -- Falling edge
                    if sclk_reg = '1' then
                        bit_count <= bit_count + 1;
                    end if;

                    -- Rising edge
                    if sclk_reg = '0' then
                        -- DB9 ... DB0
                        if bit_count >= 2 and bit_count <= 11 then
                            data_reg <= data_reg(8 downto 0) & ad_dout;
                        end if;

                        -- Finish after 16 clocks
                        if bit_count = 16 then
                            cs_reg <= '1';
                            sclk_reg <= '1';
                        end if;
                    end if;

                else
                    clk_count <= clk_count + 1;
                end if;
            end if;

        end if;
    end process;

end Behavioral;
